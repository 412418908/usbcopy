cmake_minimum_required(VERSION 3.15)
project(usbcp C)

add_executable(usbcp usbcp.c)

if(MSVC)
    add_compile_options(/W4 /utf-8)
    target_link_libraries(usbcp shell32)
else()
    # MinGW / CLang
    target_link_libraries(usbcp shell32)
    target_link_options(usbcp PRIVATE -static)
    add_compile_options(-Wall -Wextra)
endif()

install(TARGETS usbcp RUNTIME DESTINATION .)
